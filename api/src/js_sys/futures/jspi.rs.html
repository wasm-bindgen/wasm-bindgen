1//! Internal JSPI (JS Promise Integration) machinery.
2//!
3//! The public surface is a single function, `futures::jspi_block_on_promise`:
4//! suspend the current promising execution until a JS `Promise` settles. The
5//! executor side has no API at all — `spawn_local` (and everything built on
6//! it, like `future_to_promise` and async exports) checks the ambient JSPI
7//! context at spawn time and enters the task's polls through a
8//! `WebAssembly.promising` boundary when spawned from within one, so sync
9//! code reached from such a task may itself suspend. The capability is
10//! inherited transitively: a task spawned from a promising-entered poll is
11//! itself promising-entered.
12//!
13//! The runtime state is minimal by construction. The suspend bridge,
14//! `__wbindgen_jspi_suspend`, is a plain `#[wasm_bindgen(catch, suspending)]`
15//! import whose JS body is an identity function: JSPI itself awaits the
16//! returned `Promise` and resumes with the settled value, while a rejection
17//! is thrown into wasm as a `WebAssembly.JSTag` exception and marshalled to
18//! `Err` by the standard `catch + suspending` machinery (see
19//! `cli-support/src/transforms/jspi.rs`). The ambient context probe,
20//! `__wbindgen_jspi_in_context`, is rewritten in-wasm by the CLI: it reads
21//! the `__jspi_stack_base` global when JSPI instrumentation is active and is
22//! a constant `0` otherwise, so modules that never use JSPI attributes carry
23//! no JSPI machinery and keep running on engines without exnref/JSPI.
24//!
25//! On the emscripten target under `--cfg wasm_bindgen_unstable_jspi` the
26//! fibers belong to emscripten's JSPI runtime (`-sJSPI_HOOKS` /
27//! `-sREENTRANT_JSPI`): the CLI wraps the boundaries with its lifecycle hooks
28//! instead, and the context probe is a hook registered through
29//! `<emscripten/jspi.h>` that tracks whether a fiber is current.
30

32// experimental-status deprecation warning; this module is itself part of the
33// experimental JSPI surface, so silence it here.

79/// The context probe under emscripten: a lifecycle hook registered with
80/// emscripten's JSPI runtime, which reports every fiber's enter, exit,
81/// suspend and resume (for wasm-bindgen's and emscripten's own promising
82/// exports alike). The per-fiber token carries whether the fiber's host was
83/// itself in a fiber, so leaving the fiber restores the host's answer.
84#[cfg(all(target_os = "emscripten", wasm_bindgen_unstable_jspi))]
85mod emscripten {
86    use core::ffi::c_void;
87    use core::sync::atomic::{AtomicBool, Ordering::Relaxed};
88
89    const JSPI_ENTER: u32 = 1;
90    const JSPI_EXIT: u32 = 2;
91    const JSPI_SUSPEND: u32 = 4;
92    const JSPI_RESUME: u32 = 8;
93
94    type Hook = unsafe extern "C" fn(u32, *mut c_void, i32) -> *mut c_void;
95
96    extern "C" {
97        fn jspi_register(hook: Hook, mask: u32) -> i32;
98    }
99
100    // Per instance, not per thread: JSPI is rejected under `atomics`.
101    static IN_FIBER: AtomicBool = AtomicBool::new(false);
102
103    pub(crate) fn in_context() -> bool {
104        IN_FIBER.load(Relaxed)
105    }
106
107    unsafe extern "C" fn hook(event: u32, token: *mut c_void, _error: i32) -> *mut c_void {
108        match event {
109            JSPI_ENTER | JSPI_RESUME => IN_FIBER.swap(true, Relaxed) as usize as *mut c_void,
110            JSPI_SUSPEND | JSPI_EXIT => {
111                IN_FIBER.store(!token.is_null(), Relaxed);
112                token
113            }
114            _ => token,
115        }
116    }
117
118    extern "C" fn register() {
119        // -1 without `-sJSPI_HOOKS`: no events, so no fiber is ever current.
120        unsafe {
121            jspi_register(hook, JSPI_ENTER | JSPI_EXIT | JSPI_SUSPEND | JSPI_RESUME);
122        }
123    }
124
125    #[used]
126    #[link_section = ".init_array"]
127    static REGISTER: extern "C" fn() = register;
128}

130/// Suspend the current promising execution until `promise` settles.
131/// Backs `futures::jspi_block_on_promise`.
132///
133/// On return the shadow stack is guaranteed to hold the correct contents
134/// for this execution, regardless of what else ran while it was suspended:
135/// the wasm-bindgen CLI instruments every `#[wasm_bindgen(suspending)]`
136/// import with an in-wasm wrapper that evacuates the live shadow-stack
137/// region to a heap buffer before suspending and copies it back as the very
138/// first instructions after resume (see `cli-support/src/transforms/jspi.rs`).
139/// The `catch` marshalling (rejections as `Err` data) rides the same wrapper.
140#[inline(never)]
141pub(crate) fn suspend(promise: &Promise) -> Result<JsValue, JsValue> {
142    __wbindgen_jspi_suspend(promise)
143}

145// ─── Promising-entered task polls ────────────────────────────────────────────

147/// Poll-in-flight discipline for a spawned task.
148#[derive(Clone, Copy, PartialEq)]
149enum State {
150    /// Not scheduled and not being polled.
151    Idle,
152    /// A poll is queued on the microtask queue but has not started.
153    Scheduled,
154    /// A poll is in flight — executing, or suspended mid-frame.
155    Running,
156    /// A wake arrived during the in-flight poll: re-poll before going idle.
157    RunningWoken,
158}
159
160struct SpawnedTask {
161    /// `None` once the future has completed (late wakes are no-ops).
162    future: RefCell<Option<Pin<Box<dyn Future<Output = ()>>>>>,
163    state: Cell<State>,
164}
165
166impl SpawnedTask {
167    /// Schedule a poll, consuming one strong reference into the intrinsic.
168    fn schedule(this: Rc<SpawnedTask>) {
169        this.state.set(State::Scheduled);
170        __wbindgen_jspi_spawn_poll(Rc::into_raw(this) as u32);
171    }
172
173    fn wake(this: &Rc<SpawnedTask>) {
174        match this.state.get() {
175            // In flight (possibly suspended): flag a re-poll. The poll loop
176            // consumes the flag when the in-flight poll returns, so the
177            // future is never entered reentrantly.
178            State::Running => this.state.set(State::RunningWoken),
179            State::RunningWoken | State::Scheduled => {}
180            State::Idle => {
181                if this.future.borrow().is_some() {
182                    SpawnedTask::schedule(Rc::clone(this));
183                }
184            }
185        }
186    }
187
188    fn poll(this: &Rc<SpawnedTask>) {
189        loop {
190            this.state.set(State::Running);
191            let waker = task_waker(Rc::clone(this));
192            let mut cx = Context::from_waker(&waker);
193            let done = {
194                let mut slot = this.future.borrow_mut();
195                let Some(future) = slot.as_mut() else { return };
196                // The poll may suspend; `slot` stays mutably borrowed across
197                // the suspension, which is sound because every other path to
198                // the future first checks `state` (`Running`) and never
199                // touches the `RefCell`.
200                match future.as_mut().poll(&mut cx) {
201                    Poll::Ready(()) => {
202                        *slot = None;
203                        true
204                    }
205                    Poll::Pending => false,
206                }
207            };
208            if done {
209                this.state.set(State::Idle);
210                return;
211            }
212            match this.state.replace(State::Idle) {
213                // Woken while the poll was in flight: re-poll.
214                State::RunningWoken => continue,
215                _ => return,
216            }
217        }
218    }
219}

221/// A `Waker` over `Rc<SpawnedTask>`. As in `task::singlethread`, `Waker`'s
222/// nominal `Send + Sync` demand is safely ignored: JSPI is single-threaded
223/// (rejected under the `atomics` feature), so the lie is confined to the
224/// vtable.
225fn task_waker(task: Rc<SpawnedTask>) -> Waker {
226    unsafe fn raw_clone(ptr: *const ()) -> RawWaker {
227        let rc = ManuallyDrop::new(Rc::from_raw(ptr as *const SpawnedTask));
228        RawWaker::new(Rc::into_raw(Rc::clone(&rc)) as *const (), &VTABLE)
229    }
230
231    unsafe fn raw_wake(ptr: *const ()) {
232        let rc = Rc::from_raw(ptr as *const SpawnedTask);
233        SpawnedTask::wake(&rc);
234    }
235
236    unsafe fn raw_wake_by_ref(ptr: *const ()) {
237        let rc = ManuallyDrop::new(Rc::from_raw(ptr as *const SpawnedTask));
238        SpawnedTask::wake(&rc);
239    }
240
241    unsafe fn raw_drop(ptr: *const ()) {
242        drop(Rc::from_raw(ptr as *const SpawnedTask));
243    }
244
245    static VTABLE: RawWakerVTable =
246        RawWakerVTable::new(raw_clone, raw_wake, raw_wake_by_ref, raw_drop);
247
248    unsafe { Waker::from_raw(RawWaker::new(Rc::into_raw(task) as *const (), &VTABLE)) }
249}

251/// The promising-entered poll trampoline. Each call runs one task's poll
252/// loop on a fresh suspendable stack. Not public API.
253///
254/// A raw wasm export rather than a `#[wasm_bindgen(jspi)]` one: the CLI
255/// special-cases it by name — in a module with `#[wasm_bindgen(jspi)]`
256/// exports it receives the same in-wasm wrapper as a jspi export and is
257/// invoked through `WebAssembly.promising` by the spawn intrinsics' JS
258/// shims; in a module without them the ambient context is constant-false,
259/// so the export is deleted and the spawn intrinsics are stubbed out,
260/// leaving no JSPI machinery at all.
261///
262/// `extern "C-unwind"`: an unwind must be able to escape the poll — a panic
263/// under panic=unwind, or a rethrown rejection of a non-`catch` suspending
264/// import — to reject the promising call's promise. Plain `extern "C"` is
265/// nounwind, whose `panic_cannot_unwind` guard would abort the whole
266/// instance instead of abandoning the one task.
267#[no_mangle]
268pub extern "C-unwind" fn __wbg_jspi_task_poll(task: u32) {
269    let task = unsafe { Rc::from_raw(task as *const SpawnedTask) };
270    SpawnedTask::poll(&task);
271}

273/// Runs a `Future<Output = ()>` with every poll entered through a
274/// `WebAssembly.promising` boundary, so the task's whole call tree may
275/// suspend. Backs `spawn_local` when called within a JSPI context —
276/// including the rooted activation of a `#[wasm_bindgen(jspi)] async fn`
277/// export.
278///
279/// A suspension parks only this task's poll; the microtask queue, other
280/// tasks, and the event loop continue. A wake arriving while the poll is
281/// suspended re-polls after the suspended poll returns. The task is leaked
282/// if its future never completes, exactly like `spawn_local`.
283pub(crate) fn spawn_promising<F>(future: F)
284where
285    F: Future<Output = ()> + 'static,
286{
287    let task = Rc::new(SpawnedTask {
288        future: RefCell::new(Some(Box::pin(future))),
289        state: Cell::new(State::Scheduled),
290    });
291    __wbindgen_jspi_spawn_first(Rc::into_raw(task) as u32);
