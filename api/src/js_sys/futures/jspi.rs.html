1//! Internal JSPI (JS Promise Integration) machinery.
2//!
3//! The public surface is a single function, `futures::jspi_block_on_promise`:
4//! suspend the current promising execution until a JS `Promise` settles. The
5//! executor side has no API at all — `spawn_local` (and everything built on
6//! it, like `future_to_promise` and async exports) checks the ambient JSPI
7//! context at spawn time and enters the task's polls through a
8//! `WebAssembly.promising` boundary when spawned from within one, so sync
9//! code reached from such a task may itself suspend. The capability is
10//! inherited transitively: a task spawned from a promising-entered poll is
11//! itself promising-entered.
12//!
13//! The runtime state is minimal by construction. The suspend bridge,
14//! `__wbindgen_jspi_suspend`, is a plain `#[wasm_bindgen(catch, suspending)]`
15//! import whose JS body is an identity function: JSPI itself awaits the
16//! returned `Promise` and resumes with the settled value, while a rejection
17//! is thrown into wasm as a `WebAssembly.JSTag` exception and marshalled to
18//! `Err` by the standard `catch + suspending` machinery (see
19//! `cli-support/src/transforms/jspi.rs`). The ambient context probe,
20//! `__wbindgen_jspi_in_context`, is rewritten in-wasm by the CLI: it reads
21//! the `__jspi_stack_base` global when JSPI instrumentation is active and is
22//! a constant `0` otherwise, so modules that never use JSPI attributes carry
23//! no JSPI machinery and keep running on engines without exnref/JSPI.
24
25// The `suspending` attribute on the internal bridge import generates an
26// experimental-status deprecation warning; this module is itself part of the
27// experimental JSPI surface, so silence it here.
28#![allow(deprecated)]

40#[wasm_bindgen(raw_module = "__wbindgen_placeholder__")]
41extern "C" {
42    #[wasm_bindgen(catch, suspending)]
43    fn __wbindgen_jspi_suspend(promise: &Promise) -> Result<JsValue, JsValue>;

45    /// Whether a JSPI context is on the stack: a `#[wasm_bindgen(jspi)]`
46    /// export or a promising-entered poll. Rewritten in-wasm by the CLI to
47    /// read the `__jspi_stack_base` global (constant `0` in modules without
48    /// JSPI instrumentation); no JS shim is ever emitted.
49    fn __wbindgen_jspi_in_context() -> u32;

51    /// Schedules a task poll on the microtask queue, entered through the
52    /// promising-wrapped `__wbg_jspi_task_poll` export so the poll runs

54    /// reference, reclaimed by the trampoline.
55    fn __wbindgen_jspi_spawn_poll(task: u32);
56
57    /// The same operation, called only from the initial schedule in
58    /// [`spawn_promising`].
59    fn __wbindgen_jspi_spawn_first(task: u32);
60}

62/// Whether the caller is executing within a JSPI context, i.e. whether a
63/// spawned task's polls should be promising-entered.
64pub(crate) fn in_context() -> bool {
65    __wbindgen_jspi_in_context() != 0
66}

68/// Suspend the current promising execution until `promise` settles.
69/// Backs `futures::jspi_block_on_promise`.
70///
71/// On return the shadow stack is guaranteed to hold the correct contents
72/// for this execution, regardless of what else ran while it was suspended:
73/// the wasm-bindgen CLI instruments every `#[wasm_bindgen(suspending)]`
74/// import with an in-wasm wrapper that evacuates the live shadow-stack
75/// region to a heap buffer before suspending and copies it back as the very
76/// first instructions after resume (see `cli-support/src/transforms/jspi.rs`).
77/// The `catch` marshalling (rejections as `Err` data) rides the same wrapper.
78#[inline(never)]
79pub(crate) fn suspend(promise: &Promise) -> Result<JsValue, JsValue> {
80    __wbindgen_jspi_suspend(promise)
81}

85/// Poll-in-flight discipline for a spawned task.
86#[derive(Clone, Copy, PartialEq)]
87enum State {
88    /// Not scheduled and not being polled.
89    Idle,
90    /// A poll is queued on the microtask queue but has not started.
91    Scheduled,
92    /// A poll is in flight — executing, or suspended mid-frame.
93    Running,
94    /// A wake arrived during the in-flight poll: re-poll before going idle.
95    RunningWoken,
96}
97
98struct SpawnedTask {
99    /// `None` once the future has completed (late wakes are no-ops).
100    future: RefCell<Option<Pin<Box<dyn Future<Output = ()>>>>>,
101    state: Cell<State>,
102}
103
104impl SpawnedTask {
105    /// Schedule a poll, consuming one strong reference into the intrinsic.
106    fn schedule(this: Rc<SpawnedTask>) {
107        this.state.set(State::Scheduled);
108        __wbindgen_jspi_spawn_poll(Rc::into_raw(this) as u32);
109    }
110
111    fn wake(this: &Rc<SpawnedTask>) {
112        match this.state.get() {
113            // In flight (possibly suspended): flag a re-poll. The poll loop
114            // consumes the flag when the in-flight poll returns, so the
115            // future is never entered reentrantly.
116            State::Running => this.state.set(State::RunningWoken),
117            State::RunningWoken | State::Scheduled => {}
118            State::Idle => {
119                if this.future.borrow().is_some() {
120                    SpawnedTask::schedule(Rc::clone(this));
121                }
122            }
123        }
124    }
125
126    fn poll(this: &Rc<SpawnedTask>) {
127        loop {
128            this.state.set(State::Running);
129            let waker = task_waker(Rc::clone(this));
130            let mut cx = Context::from_waker(&waker);
131            let done = {
132                let mut slot = this.future.borrow_mut();
133                let Some(future) = slot.as_mut() else { return };
134                // The poll may suspend; `slot` stays mutably borrowed across
135                // the suspension, which is sound because every other path to
136                // the future first checks `state` (`Running`) and never
137                // touches the `RefCell`.
138                match future.as_mut().poll(&mut cx) {
139                    Poll::Ready(()) => {
140                        *slot = None;
141                        true
142                    }
143                    Poll::Pending => false,
144                }
145            };
146            if done {
147                this.state.set(State::Idle);
148                return;
149            }
150            match this.state.replace(State::Idle) {
151                // Woken while the poll was in flight: re-poll.
152                State::RunningWoken => continue,
153                _ => return,
154            }
155        }
156    }
157}

159/// A `Waker` over `Rc<SpawnedTask>`. As in `task::singlethread`, `Waker`'s
160/// nominal `Send + Sync` demand is safely ignored: JSPI is single-threaded
161/// (rejected under the `atomics` feature), so the lie is confined to the
162/// vtable.
163fn task_waker(task: Rc<SpawnedTask>) -> Waker {
164    unsafe fn raw_clone(ptr: *const ()) -> RawWaker {
165        let rc = ManuallyDrop::new(Rc::from_raw(ptr as *const SpawnedTask));
166        RawWaker::new(Rc::into_raw(Rc::clone(&rc)) as *const (), &VTABLE)
167    }
168
169    unsafe fn raw_wake(ptr: *const ()) {
170        let rc = Rc::from_raw(ptr as *const SpawnedTask);
171        SpawnedTask::wake(&rc);
172    }
173
174    unsafe fn raw_wake_by_ref(ptr: *const ()) {
175        let rc = ManuallyDrop::new(Rc::from_raw(ptr as *const SpawnedTask));
176        SpawnedTask::wake(&rc);
177    }
178
179    unsafe fn raw_drop(ptr: *const ()) {
180        drop(Rc::from_raw(ptr as *const SpawnedTask));
181    }
182
183    static VTABLE: RawWakerVTable =
184        RawWakerVTable::new(raw_clone, raw_wake, raw_wake_by_ref, raw_drop);
185
186    unsafe { Waker::from_raw(RawWaker::new(Rc::into_raw(task) as *const (), &VTABLE)) }
187}

189/// The promising-entered poll trampoline. Each call runs one task's poll
190/// loop on a fresh suspendable stack. Not public API.
191///
192/// A raw wasm export rather than a `#[wasm_bindgen(jspi)]` one: the CLI
193/// special-cases it by name — in a module with `#[wasm_bindgen(jspi)]`
194/// exports it receives the same in-wasm wrapper as a jspi export and is
195/// invoked through `WebAssembly.promising` by the spawn intrinsics' JS
196/// shims; in a module without them the ambient context is constant-false,
197/// so the export is deleted and the spawn intrinsics are stubbed out,
198/// leaving no JSPI machinery at all.
199///
200/// `extern "C-unwind"`: an unwind must be able to escape the poll — a panic
201/// under panic=unwind, or a rethrown rejection of a non-`catch` suspending
202/// import — to reject the promising call's promise. Plain `extern "C"` is
203/// nounwind, whose `panic_cannot_unwind` guard would abort the whole
204/// instance instead of abandoning the one task.
205#[no_mangle]
206pub extern "C-unwind" fn __wbg_jspi_task_poll(task: u32) {
207    let task = unsafe { Rc::from_raw(task as *const SpawnedTask) };
208    SpawnedTask::poll(&task);
209}

211/// Runs a `Future<Output = ()>` with every poll entered through a
212/// `WebAssembly.promising` boundary, so the task's whole call tree may
213/// suspend. Backs `spawn_local` when called within a JSPI context —
214/// including the rooted activation of a `#[wasm_bindgen(jspi)] async fn`
215/// export.
216///
217/// A suspension parks only this task's poll; the microtask queue, other
218/// tasks, and the event loop continue. A wake arriving while the poll is
219/// suspended re-polls after the suspended poll returns. The task is leaked
220/// if its future never completes, exactly like `spawn_local`.
221pub(crate) fn spawn_promising<F>(future: F)
222where
223    F: Future<Output = ()> + 'static,
224{
225    let task = Rc::new(SpawnedTask {
226        future: RefCell::new(Some(Box::pin(future))),
227        state: Cell::new(State::Scheduled),
228    });
229    __wbindgen_jspi_spawn_first(Rc::into_raw(task) as u32);
230}
